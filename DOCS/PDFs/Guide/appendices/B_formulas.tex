\chapter{Formula Sheet}
\label{app:formulas}

Every formula below follows the project conventions (Hamilton quaternions, local perturbations) and was verified numerically. Let
$\theta = \|\phi\|$.

\section*{Cross products}
\begin{gather*}
  \sx{a}b = a\times b = -\sx{b}a, \qquad \sx{a}\T = -\sx{a}, \qquad \sx{a}^2 = aa\T - \|a\|^2I, \qquad \sx{a}^3 = -\|a\|^2\sx{a}, \\
  \sx{a}\sx{b} = ba\T - (a\T b)I, \qquad \sx{a\times b} = \sx{a}\sx{b} - \sx{b}\sx{a}, \qquad \sx{Ra} = R\sx{a}R\T .
\end{gather*}

\section*{Exponential and logarithm}
\begin{gather*}
  \Exp(\phi) = I + \frac{\sin\theta}{\theta}\sx{\phi} + \frac{1 - \cos\theta}{\theta^2}\sx{\phi}^2, \qquad
  \Exp_q(\phi) = \Big[\cos\tfrac\theta2,\ \tfrac{\sin(\theta/2)}{\theta}\phi\Big], \\
  \theta = \atantwo\big(\tfrac12\|(R - R\T)\veeop\|,\ \tfrac12(\tr R - 1)\big), \qquad
  \Log(R) = \frac{\theta}{2\sin\theta}(R - R\T)\veeop \quad (0 < \theta < \pi), \\
  \text{near }\pi:\ uu\T = \frac{\tfrac12(R + R\T) - \cos\theta\,I}{1 - \cos\theta}, \qquad
  \Log_q(q) = 2\atantwo(\|v\|, w)\frac{v}{\|v\|}\ \text{ after } w \ge 0 .
\end{gather*}

\section*{Jacobians of \texorpdfstring{$\SO$}{SO(3)}}
\begin{gather*}
  \Jr(\phi) = I - B\,\sx{\phi} + C\,\sx{\phi}^2, \qquad \Jrinv(\phi) = I + \tfrac12\sx{\phi} + D\,\sx{\phi}^2, \\
  \Jl(\phi) = \Jr(-\phi) = \Jr(\phi)\T = \Exp(\phi)\Jr(\phi), \qquad \Jr(\phi) = \int_0^1\Exp(-s\phi)\,\dd s, \\
  \Exp(\phi + \delta) \approx \Exp(\phi)\Exp(\Jr(\phi)\delta), \qquad
  \Log(\Exp(\phi)\Exp(\delta)) \approx \phi + \Jrinv(\phi)\delta, \\
  R\Exp(\phi)R\T = \Exp(R\phi), \qquad \Log(\Exp(a)\Exp(b)) = a + b + \tfrac12 a\times b + \cdots
\end{gather*}

\section*{Quaternions}
\begin{gather*}
  p\qprod q = \big[p_wq_w - p_v\T q_v,\ \ p_wq_v + q_wp_v + p_v\times q_v\big], \qquad
  R(q) = (w^2 - v\T v)I + 2vv\T + 2w\sx{v}, \\
  \dot q = \tfrac12 q\qprod[0, \omega_b], \qquad \dot R = R\sx{\omega_b}, \qquad R(p\qprod q) = R(p)R(q).
\end{gather*}

\section*{Linear algebra and least squares}
\begin{gather*}
  (A + UCV)\inv = A\inv - A\inv U\big(C\inv + VA\inv U\big)\inv VA\inv \quad \text{(Woodbury)}, \\
  \begin{bmatrix} A & B \\ C & D\end{bmatrix}\inv\!: \ \text{Schur complement } S_A = D - CA\inv B, \\
  \hat x_{\mathrm{LS}} = (A\T A)\inv A\T b = R\inv Q_1\T b \ \ (A = Q_1R), \\
  \nullsp(A\T) = \range(Q_2) \ \ \Big(A = [Q_1\ Q_2]\begin{bmatrix} R \\ 0\end{bmatrix}\Big), \qquad
  \kappa_2(A) = \sigma_{\max}/\sigma_{\min}, \\
  \min_{\|x\| = 1}\|Ax\| = \sigma_{\min}, \text{ attained at the right singular vector of } \sigma_{\min} .
\end{gather*}

\section*{Gaussians}
\begin{gather*}
  \N(x; \mu, \Sigma) = \frac{\exp\big(-\tfrac12(x - \mu)\T\Sigma\inv(x - \mu)\big)}{\sqrt{(2\pi)^n\det\Sigma}}, \qquad
  Ax + b \sim \N(A\mu + b,\ A\Sigma A\T), \\
  \mu_{a|b} = \mu_a + \Sigma_{ab}\Sigma_{bb}\inv(b - \mu_b), \qquad \Sigma_{a|b} = \Sigma_{aa} - \Sigma_{ab}\Sigma_{bb}\inv\Sigma_{ba}, \\
  (x - \mu)\T\Sigma\inv(x - \mu) \sim \chi^2_n, \qquad
  \text{fusion: } \Sigma\inv = \Sigma_1\inv + \Sigma_2\inv,\ \ \Sigma\inv\mu = \Sigma_1\inv\mu_1 + \Sigma_2\inv\mu_2 .
\end{gather*}

\section*{Random processes and discretization}
\begin{gather*}
  \dot P = FP + PF\T + GQ_cG\T \quad \text{(Lyapunov)}, \qquad
  Q_d = \int_0^\Dt e^{Fs}GQ_cG\T e^{F\T s}\,\dd s, \\
  \exp\!\Big(\begin{bmatrix} -F & GQ_cG\T \\ 0 & F\T\end{bmatrix}\Dt\Big) = \begin{bmatrix} C_{11} & C_{12} \\ 0 & C_{22}\end{bmatrix}:\quad
  \Phi = C_{22}\T,\ \ Q_d = \Phi C_{12} \quad \text{(Van Loan)} .
\end{gather*}

\section*{Kalman filter}
\begin{gather*}
  P \gets \Phi P\Phi\T + Q_d, \qquad \Phi \approx I + F_c\Dt, \qquad Q_d \approx G_cQ_cG_c\T\Dt, \\
  r = z\bminus h(\hat x), \quad H = \tfrac{\partial}{\partial\delta x}\big[h(\hat x\bplus\delta x)\bminus h(\hat x)\big]_0, \quad
  S = HPH\T + R_z, \quad K = PH\T S\inv, \\
  P \gets (I - KH)P(I - KH)\T + KR_zK\T, \qquad \hat x \gets \hat x\bplus Kr, \qquad P \gets GPG\T,\ G_\theta = I - \tfrac12\sx{\delta\hat\theta}.
\end{gather*}

\section*{Unscented transform}
\begin{gather*}
  \lambda = \alpha^2(n + \kappa) - n, \qquad \chi_0 = \mu, \quad \chi_{i}, \chi_{n+i} = \mu \pm \sqrt{n + \lambda}\,\ell_i \ \ (P = LL\T), \\
  W_0^{(m)} = \frac{\lambda}{n + \lambda}, \quad W_0^{(c)} = \frac{\lambda}{n + \lambda} + 1 - \alpha^2 + \beta, \quad W_i^{(m)} = W_i^{(c)} = \frac{1}{2(n + \lambda)} .
\end{gather*}

\section*{Nonlinear least squares}
\begin{gather*}
  (J\T WJ + \lambda I)\,\delta = -J\T Wr, \qquad x \gets x\bplus\delta, \\
  \lambda = 0\text{: Gauss--Newton}, \qquad \lambda > 0\text{: Levenberg--Marquardt}, \\
  \text{Huber weight } w(s) = \min(1, c/s), \qquad \text{Cauchy weight } w(s) = 1/(1 + s^2/c^2), \\
  \begin{bmatrix} A & B \\ B\T & C\end{bmatrix}\begin{bmatrix}\delta c \\ \delta l\end{bmatrix} = \begin{bmatrix} g_c \\ g_l\end{bmatrix}
  \ \ \Rightarrow\ \ (A - BC\inv B\T)\,\delta c = g_c - BC\inv g_l \ \ \text{(reduced camera system)} .
\end{gather*}

\section*{Strapdown propagation with constant inputs (\texorpdfstring{$\phi = \hat\omega\Dt$}{phi = omega dt})}
\begin{gather*}
  R_{k+1} = R_k\Exp(\phi), \qquad v_{k+1} = v_k + g\Dt + R_k\Dt\Jl(\phi)\hat a, \\
  p_{k+1} = p_k + v_k\Dt + \tfrac12g\Dt^2 + R_k\Dt^2N(\phi)\hat a, \\
  N(\phi) = \tfrac12I + \frac{\theta - \sin\theta}{\theta^3}\sx{\phi} + \frac{\theta^2/2 + \cos\theta - 1}{\theta^4}\sx{\phi}^2 .
\end{gather*}
Bortz equation and the two-sample coning correction (increments $\alpha_1, \alpha_2$ over the two halves of a step):
\begin{gather*}
  \dot\phi = \Jrinv(\phi)\,\omega = \omega + \tfrac12\phi\times\omega + D(\theta)\,\phi\times(\phi\times\omega), \qquad
  \phi \approx \alpha_1 + \alpha_2 + \tfrac23\,\alpha_1\times\alpha_2 .
\end{gather*}

\section*{Noise}
\begin{gather*}
  \sigma_d = \sigma/\sqrt\Dt\ \ \text{(white, per sample)}, \qquad \sigma_{b,d} = \sigma_b\sqrt\Dt\ \ \text{(random walk, per step)}, \\
  \Var(m\text{-fold integral of white noise}) = \sigma^2\frac{t^{2m-1}}{(2m-1)((m-1)!)^2}, \\
  1\ ^\circ/\sqrt{\mathrm h} = 2.909\times10^{-4}~\mathrm{rad/s/\sqrt{Hz}} .
\end{gather*}

\section*{Consistency and alignment}
\begin{gather*}
  \epsilon = e\T P\inv e \sim \chi^2_n \ \text{(NEES)}, \qquad \nu = r\T S\inv r \sim \chi^2_m \ \text{(NIS)}, \\
  \bar\epsilon \in \Big[\frac{\chi^2_{0.025}(Nn)}{N},\ \frac{\chi^2_{0.975}(Nn)}{N}\Big] \ \ \text{(average over } N\text{ runs, 95\%)}, \\
  \Sigma_{ab} = \tfrac1N\textstyle\sum_i a_ib_i\T = UDV\T:\quad R^\star = U\diag(1, 1, \det(UV\T))V\T, \quad t^\star = \mu - R^\star\hat\mu \quad \text{(Umeyama)} .
\end{gather*}

\section*{Geometry}
\begin{gather*}
  \bar x_2\T\sx{t}R\bar x_1 = 0, \qquad \frac{\partial\pi}{\partial p_c} = \frac1Z\begin{bmatrix} 1 & 0 & -X/Z \\ 0 & 1 & -Y/Z\end{bmatrix}, \\
  \sigma_Z \approx \sqrt2\frac{Z^2}{b}\sigma_{uv}, \qquad N_{\mathrm{RANSAC}} = \frac{\log(1-p)}{\log(1 - w^s)} .
\end{gather*}

\section*{MSCKF}
\begin{gather*}
  r_j \approx H_{x,j}\,\delta x + H_{f,j}\,\delta p_f + n_j, \qquad A_j\T H_{f,j} = 0,\ \ A_j\T A_j = I, \\
  r_{o,j} = A_j\T r_j,\quad H_{o,j} = A_j\T H_{x,j},\quad \Cov(A_j\T n_j) = \sigma^2 I, \\
  \dim r_{o,j} = 2M - 3 \ \text{(space)},\ \ M - 2 \ \text{(plane)}, \\
  \text{gate: } r_{o,j}\T(H_{o,j}PH_{o,j}\T + \sigma^2I)\inv r_{o,j} < \chi^2_{0.95}(\dim r_{o,j}), \\
  H_o = Q_1T_H \ \Rightarrow\ (r_n, H_n) = (Q_1\T r_o, T_H) \ \text{(compression)}, \\
  P \gets JPJ\T,\ \ J = \begin{bmatrix} I \\ J_g\end{bmatrix} \ \text{(augmentation)} .
\end{gather*}

\section*{Planar models}
\begin{gather*}
  \Omega = \begin{bmatrix} 0 & -1 \\ 1 & 0\end{bmatrix}, \qquad R(\psi) = \cos\psi\,I + \sin\psi\,\Omega, \qquad R\Omega = \Omega R, \\
  R(\psi + \delta)\T \approx (I - \delta\,\Omega)R(\psi)\T, \qquad z = y/x: \quad J_\pi = \tfrac1x\begin{bmatrix} -z & 1\end{bmatrix}, \\
  \frac{\partial z}{\partial\delta p_i} = -J_\pi R_i\T, \qquad
  \frac{\partial z}{\partial\delta\psi_i} = -(1 + z^2), \qquad \frac{\partial z}{\partial p_f} = J_\pi R_i\T, \\
  \Phi_k = \begin{bmatrix} I & \Omega(\hat p_{k+1} - \hat p_k) \\ 0 & 1\end{bmatrix}, \qquad
  N = \Big[\begin{bmatrix} I_2 \\ 0 \\ I_2 \\ \vdots\end{bmatrix}\ \begin{bmatrix} \Omega\hat p \\ 1 \\ \Omega\hat p_1 \\ \vdots\end{bmatrix}\Big].
\end{gather*}

\section*{Observability}
\begin{gather*}
  \mathcal{O} = \begin{bmatrix} H_1 \\ H_2\Phi_1 \\ \vdots \\ H_k\Phi_{k-1}\cdots\Phi_1\end{bmatrix}, \qquad
  L_f\varphi = \nabla\varphi\cdot f, \\
  N_{\text{yaw}} = \big(-\sx{\hat p}e_z,\ -\sx{\hat v}e_z,\ \hat R\T e_z,\ 0,\ 0,\ -\sx{\hat p_i}e_z,\ \hat R_i\T e_z\big), \\
  H^* = H\big(I - N(N\T N)\inv N\T\big), \qquad \Phi^* = \Phi + (N_{k+1} - \Phi N_k)(N_k\T N_k)\inv N_k\T \quad \text{(OC-EKF)} .
\end{gather*}
